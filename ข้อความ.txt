<script>

let price = 10

let items = [
{name:"ลูกชิ้นปลา",stock:50,sold:0},
{name:"ไส้กรอกแดง",stock:50,sold:0},
{name:"ชีสบอล",stock:50,sold:0}
]

let salesHistory = []

function renderItems(){

let html=""

items.forEach((item,i)=>{

html+=`

<div class="item">

<div>

${item.name}

<br>

เหลือ ${item.stock}

</div>

<div>

<button onclick="sell(${i})">+1</button>

</div>

</div>

`

})

document.getElementById("items").innerHTML=html

updateTotals()

}

function sell(i){

if(items[i].stock<=0){

alert("ของหมด")

return

}

items[i].stock--

items[i].sold++

renderItems()

}

function updateTotals(){

let sticks=0

items.forEach(i=>sticks+=i.sold)

document.getElementById("totalSticks").innerText=

sticks+" ไม้"

document.getElementById("totalMoney").innerText=

"฿"+sticks*price

}

function closeShop(){

let sticks=0

items.forEach(i=>sticks+=i.sold)

let expected=sticks*price

let cash=parseFloat(

document.getElementById("cash").value||0)

let transfer=parseFloat(

document.getElementById("transfer").value||0)

let actual=cash+transfer

let diff=actual-expected

document.getElementById("summary").innerHTML=

`

ยอดขาย ${sticks} ไม้<br>

ควรได้ ฿${expected}<br>

ได้เงินจริง ฿${actual}<br>

ขาด/เกิน ${diff}

`

let branch=document.getElementById("branch").value

salesHistory.push({

branch:branch,

money:expected,

date:new Date()

})

updateBranchRank()

updateMonthly()

}

function updateBranchRank(){

let totals={}

salesHistory.forEach(s=>{

if(!totals[s.branch]) totals[s.branch]=0

totals[s.branch]+=s.money

})

let arr=Object.entries(totals)

arr.sort((a,b)=>b[1]-a[1])

let html=""

arr.forEach((b,i)=>{

html+=`${i+1}. ${b[0]} - ฿${b[1]}<br>`

})

document.getElementById("branchRank").innerHTML=html

}

function updateMonthly(){

let month=new Date().getMonth()

let total=0

salesHistory.forEach(s=>{

if(new Date(s.date).getMonth()==month)

total+=s.money

})

document.getElementById("monthSales").innerText=

"ยอดเดือนนี้ ฿"+total

}

function addItem(){

let name=prompt("ชื่อสินค้า")

if(!name) return

items.push({name:name,stock:50,sold:0})

renderItems()

}

renderItems()
